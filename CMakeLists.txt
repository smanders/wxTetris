cmake_minimum_required(VERSION 3.31)
set(CMAKE_MODULE_PATH ${CMAKE_CURRENT_LIST_DIR}/.devcontainer/cmake)
include(preproject)
project(wxTetris VERSION 0.9.0)
include(xptoplevel)
set(exe_srcs src/App.cpp src/Tetris.cpp src/Tetris.hpp LICENSE)
add_executable(${PROJECT_NAME} WIN32 ${exe_srcs})
xpFindPkg(PKGS exdlpro wxwidgets)
target_link_libraries(${PROJECT_NAME} PRIVATE wx::core)
install(TARGETS ${PROJECT_NAME})
set(CPACK_GENERATOR TXZ)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  set(CPACK_PACKAGE_FILE_NAME "${CMAKE_PROJECT_NAME}-${CMAKE_PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-arm64")
endif()
include(CPack)
xpSourceListAppend("${exe_srcs}")
